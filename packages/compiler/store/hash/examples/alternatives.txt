type T = | A(Int, Int) | B(Int, Int) | C

fn small(n: Int) -> Int = match n { 1 | 2 | 3 -> 10, 4 | (5 | 6) -> 20, _ -> 0 }

fn swapped(t: T) -> Int = match t { A(x, y) | B(y, x) -> x * 10 + y, C -> 0 }

fn guarded(t: T) -> Int = match t { A(x, y) | B(y, x) if x > y -> x, _ -> 0 - 1 }

fn nested(ts: List<T>) -> Int =
  match ts { [A(x, _) | B(_, x), ..rest] | [C, A(x, _), ..rest] -> x + rest.len(), _ -> 0 }

fn fields(p: { a: Int, b: Int }) -> Int =
  match p { { a: 0, b: v } | { a: v, b: 0 } -> v, _ -> 0 - 1 }

fn left(t: T) -> Int = {
  let A(n, _) | B(_, n) = t else { 0 - 5 };
  n
}

fn both(r: Result<Int, Int>) -> Int = {
  let Ok(v) | Err(v) = r;
  v
}

fn main() -> Int =
  small(2) + small(5) + swapped(B(1, 2)) + guarded(A(5, 1)) + nested([C, A(3, 0), C, C])
    + fields({ a: 6, b: 0 })
    + left(C)
    + both(Err(2))
