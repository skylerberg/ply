effect note {
  write left(n: Int) -> Unit
}

fn weighed<[r]>(h: Scoped<Int, [r]>, by: Int) -> Int / {scoped.read[r]} = h.get() * by

fn scoped_doubled(n: Int) -> Int / {note.write} =
  with_scope[slot](n + 1, |held: Int| note.left(held)) { h -> weighed(h, 2) }

fn nested(n: Int) -> Int / {note.write} =
  with_scope[outer](n, |held: Int| note.left(held)) { a ->
    with_scope[inner]([2, a.get()], |held: List<Int>| note.left(held.len())) { b ->
      b.get().map(|x: Int| x + a.get()).len() + a.get()
    }
  }

fn rejected(n: Int) -> Int / {note.write} =
  handle {
    with_scope[slot](n, |held: Int| note.left(held)) { h -> b"ab".at(2 + h.get()) }
  } with {
    abort.raise(m) -> 0 - 1,
  }

fn main() -> Int =
  handle {
    scoped_doubled(2) + nested(5) + rejected(3)
  } with {
    var seen = 0,
    note.left(held) -> {
      seen = seen * 10 + held;
    },
    return result -> result * 100000 + seen,
  }
