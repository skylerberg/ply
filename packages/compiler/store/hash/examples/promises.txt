effect log {
  write emit(n: Int) -> Unit
}

fn first(xs: List<Int>) -> Int = {
  let [x, ..] = xs else { 0 - 1 };
  x
}

fn list_if_positive(n: Int, xs: List<Int>) -> List<Int>
  returns xs
= if n > 0 { xs } else { [] }

fn made(n: Int) -> List<Int>
  returns fresh
= [n, n]

fn noted<| e>(n: Int, f: (Int) -> Int / e) -> Int / {log.write bounded | e bounded} = {
  log.emit(n);
  f(n)
}

fn digit(b: Int) -> Int = match b { b'0'..=b'9' -> b - b'0', _ -> 0 - 1 }

fn main() -> Int =
  first(made(7)) + first([]) + list_if_positive(1, [1, 2]).len() + digit(b'4') * 10
    + (handle { noted(3, |n: Int| n * 100) } with { log.emit(n) -> () })
