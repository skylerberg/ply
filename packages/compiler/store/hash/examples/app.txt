pub fn exprs(x: Int, pt: Point, t: Tree, xs: List<Int>, bag: Bag, c: Cart) -> Int / {net.write[conn], abort.raise} = {
  let neg = -x;
  let sum = x + 1;
  let same = xs == xs;
  let f = |y: Int| y + x;
  let called = f(x);
  let relayed = relay[conn](x);
  let chosen = if x > 0 { 1 } else { 2 };
  let matched = match t { Leaf -> 0, Node(_, n) | Other(n, _) -> n, _ -> 1 };
  let made = { at: x, up: 1 };
  let moved = { ..pt, at: 1 };

pub fn stmts(t: Tree, xs: List<Int>, c: Bool) -> Int / {abort.raise} = {
  let Node(_, n) | Other(n, _) = t else { 0 - 1 };
  let [h, ..] = xs else { if c { n } else { 0 } };
  let (a, b): (Int, Int) = (n, h);
  assert(a <= b);
  a + b
}

pub fn shadowed(x: Int, b: Int) -> Int = {
  let a = b;
  let b = x + 1;
  let c = { let a = 1; a };
  let d = 1;
  let d = 2;
  a + b + c + d
}

pub fn pats(v: Int, s: String, t: Tree, o: Option<U8>, pt: Point, pair: (Int, Tree), ys: List<Int>) -> Int =
  match v { 1 -> 1, 0..=9 -> 2, -5..=-1 -> 3, 7 | 8 -> 4, n if n > 100 -> n, x -> x }
    + match s { "s" -> 1, _ -> 0 }
    + match o { Some(1u8..=9u8 | 200u8..=255u8) -> 1, _ -> 0 }
    + match pt { {at: 0, up: u} | {at: u, up: 0} -> u, {at: a, ..} -> a }
    + match pair { (a, Leaf) | (_, (Node(_, a) | Other(a, _))) -> a, _ -> 0 }
    + match ys { [] -> 0, [h, ..rest] | [_, h, ..rest] -> h, _ -> 1 }
    + match t { Node(Node(_, a), b) | Other(b, Node(_, a)) -> a - b, _ -> 0 }
