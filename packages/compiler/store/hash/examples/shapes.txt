pub type T = | A(Int, Int) | B(Int, Int) | C

pub fn pick(t: T) -> Int = match t { A(x, y) | B(y, x) -> x * 10 + y, C -> 0 }

pub transparent fn small(n: Int) -> Bool = match n { 1 | 2 | 3 -> true, _ -> false }

pub fn first(xs: List<Int>) -> Int = {
  let [x, ..] | [_, x, ..] = xs else { 0 - 1 };
  x
}
